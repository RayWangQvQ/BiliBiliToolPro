**Step 2 — Create `src/Ray.BiliBiliTool.Infrastructure/Notifications/SerilogNotificationAdapter.cs`:**

```csharp
using Microsoft.Extensions.Logging;
using Ray.BiliBiliTool.Application.Contracts.Notifications;
using Serilog;

namespace Ray.BiliBiliTool.Infrastructure.Notifications;

/// <summary>
/// Default INotificationService implementation that routes notification content
/// through the existing Serilog ILogger pipeline. All 13 sink channels continue
/// to work identically — this adapter is the zero-breakage bridge.
/// </summary>
public class SerilogNotificationAdapter : INotificationService
{
    private readonly ILogger<SerilogNotificationAdapter> _logger;

    public SerilogNotificationAdapter(ILogger<SerilogNotificationAdapter> logger)
    {
        _logger = logger;
    }

    public Task SendAsync(NotificationMessage message, string groupKey)
    {
        try
        {
            var logLevel = MapLevel(message.Level);
            var text = $"[NOTIFICATION] {message.Title}: {message.Body}";
            _logger.Log(logLevel, text);
        }
        catch (Exception ex)
        {
            LogFailure(ex);
        }

        return Task.CompletedTask;
    }

    public Task SendSummaryAsync(string title, SummaryLine[] lines, string groupKey)
    {
        try
        {
            var formatted = FormatSummary(title, lines);
            // Summary inherits the most severe level from lines
            // (default to Info since summaries are task-completion reports)
            _logger.LogInformation(formatted);
        }
        catch (Exception ex)
        {
            LogFailure(ex);
        }

        return Task.CompletedTask;
    }

    private static string FormatSummary(string title, SummaryLine[] lines)
    {
        var sb = new System.Text.StringBuilder();
        sb.Append("[NOTIFICATION] ");
        sb.AppendLine(title);
        foreach (var line in lines)
        {
            var icon = line.StatusIcon ?? "•";
            sb.AppendLine($"  {icon} {line.Label}: {line.Value}");
        }
        return sb.ToString().TrimEnd();
    }

    private static LogLevel MapLevel(NotificationLevel level) => level switch
    {
        NotificationLevel.Info => LogLevel.Information,
        NotificationLevel.Warning => LogLevel.Warning,
        NotificationLevel.Error => LogLevel.Error,
        _ => LogLevel.Information
    };

    /// <summary>
    /// Logs adapter errors through a separate path so notification sinks
    /// can filter them out (prevents circular error loops).
    /// Uses Serilog's static Log.ForContext instead of the injected ILogger
    /// to mark the origin explicitly (SERILOG-04).
    /// </summary>
    private static void LogFailure(Exception ex)
    {
        Serilog.Log
            .ForContext("NotificationOrigin", "adapter")
            .Error(ex, "SerilogNotificationAdapter failed to dispatch notification");
    }
}
```

Key design notes:
- Both methods return `Task.CompletedTask` — they're synchronous writes to ILogger. The async signature matches the port contract for future non-Serilog adapters.
- `SendAsync` maps `NotificationLevel` → `LogLevel` (Info→Information, Warning→Warning, Error→Error) per D-06.
- `SendSummaryAsync` uses `LogInformation` because summaries are task-completion reports (informational by nature).
- `FormatSummary` produces a readable multi-line message with title and indented lines.
- Error isolation: `LogFailure` uses `Serilog.Log.ForContext("NotificationOrigin", "adapter")` — sinks can filter `NotificationOrigin == "adapter"` to prevent circular dependency (Pitfall 3 / SERILOG-04).
- The adapter does NOT push `GroupPropertyKey` — that's already in `LogContext` from `BaseJob.Execute()`. The adapter just writes through ILogger and inherits the ambient context (SERILOG-03).
- No batch management — `BatchSinkManager.FlushAsync()` is called by `BaseJob` at job end. The adapter stays out of flush timing (D-02).

using FluentAssertions;
using Microsoft.Extensions.Logging;
using Ray.BiliBiliTool.Application.Contracts.Notifications;
using Ray.BiliBiliTool.Infrastructure.Notifications;

namespace InfrastructureTest;

public class SerilogNotificationAdapterTests
{
    [Fact]
    public void SendAsync_logs_notification_at_correct_level()
    {
        // Arrange
        var logger = new TestLogger<SerilogNotificationAdapter>();
        var adapter = new SerilogNotificationAdapter(logger);
        var message = new NotificationMessage("Test Title", "Test Body", NotificationLevel.Warning);

        // Act
        adapter.SendAsync(message, "group-1").Wait();

        // Assert
        logger.LastLevel.Should().Be(LogLevel.Warning);
        logger.LastMessage.Should().Contain("[NOTIFICATION] Test Title: Test Body");
    }

    [Fact]
    public void SendAsync_maps_info_level_to_information()
    {
        var logger = new TestLogger<SerilogNotificationAdapter>();
        var adapter = new SerilogNotificationAdapter(logger);
        var message = new NotificationMessage("Title", "Body", NotificationLevel.Info);

        adapter.SendAsync(message, "g1").Wait();

        logger.LastLevel.Should().Be(LogLevel.Information);
    }

    [Fact]
    public void SendAsync_maps_error_level_to_error()
    {
        var logger = new TestLogger<SerilogNotificationAdapter>();
        var adapter = new SerilogNotificationAdapter(logger);
        var message = new NotificationMessage("Title", "Body", NotificationLevel.Error);

        adapter.SendAsync(message, "g1").Wait();

        logger.LastLevel.Should().Be(LogLevel.Error);
    }

    [Fact]
    public void SendSummaryAsync_formats_title_and_lines()
    {
        var logger = new TestLogger<SerilogNotificationAdapter>();
        var adapter = new SerilogNotificationAdapter(logger);
        var lines = new[]
        {
            new SummaryLine("投币", "5 个", "✅"),
            new SummaryLine("签到", "完成")
        };

        adapter.SendSummaryAsync("Daily", lines, "g1").Wait();

        logger.LastLevel.Should().Be(LogLevel.Information);
        logger.LastMessage.Should().Contain("[NOTIFICATION] Daily");
        logger.LastMessage.Should().Contain("✅ 投币: 5 个");
        logger.LastMessage.Should().Contain("• 签到: 完成");
    }

    [Fact]
    public void SendSummaryAsync_defaults_icon_to_bullet_when_null()
    {
        var logger = new TestLogger<SerilogNotificationAdapter>();
        var adapter = new SerilogNotificationAdapter(logger);
        var lines = new[] { new SummaryLine("Key", "Value") };

        adapter.SendSummaryAsync("Title", lines, "g1").Wait();

        logger.LastMessage.Should().Contain("• Key: Value");
    }

    [Fact]
    public void SendAsync_returns_completed_task()
    {
        var logger = new TestLogger<SerilogNotificationAdapter>();
        var adapter = new SerilogNotificationAdapter(logger);
        var message = new NotificationMessage("T", "B", NotificationLevel.Info);

        var task = adapter.SendAsync(message, "g1");

        task.IsCompleted.Should().BeTrue();
    }
}

/// <summary>
/// Minimal test logger that captures the last log call for assertions.
/// </summary>
internal class TestLogger<T> : ILogger<T>
{
    public LogLevel? LastLevel { get; private set; }
    public string? LastMessage { get; private set; }
    public Exception? LastException { get; private set; }

    public IDisposable? BeginScope<TState>(TState state) where TState : notnull => null;
    public bool IsEnabled(LogLevel logLevel) => true;

    public void Log<TState>(
        LogLevel logLevel,
        EventId eventId,
        TState state,
        Exception? exception,
        Func<TState, Exception?, string> formatter)
    {
        LastLevel = logLevel;
        LastMessage = formatter(state, exception);
        LastException = exception;
    }
}

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| AppService → SerilogNotificationAdapter | Application-layer code invokes the adapter — must only go through INotificationService port, not directly reference the adapter class |
| SerilogNotificationAdapter → Serilog pipeline | Adapter writes to ILogger — Serilog sinks see the log events; batching handled by existing LogContext + BatchSinkManager |
| Adapter error logging → notification sinks | Adapter errors must NOT re-enter the notification sink pipeline (circular dependency prevention via ForContext) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-21-01 | Tampering | SerilogNotificationAdapter | mitigate | Adapter is in Infrastructure layer; ArchUnit guardrail prevents Application from directly referencing adapter class, enforcing port usage only |
| T-21-02 | Information Disclosure | Adapter error logging | mitigate | Error logs use `Log.ForContext("NotificationOrigin", "adapter")` so notification sinks filter them out — prevents error details from leaking through user-facing notification channels |
| T-21-03 | Denial of Service | Adapter logging failure | accept | Adapter catches exceptions and logs via separate path — a logging failure won't crash the host. Transient Serilog failures are self-healing. No PII in notification content. |
</threat_model>
